function doGet(e) { var message = e.parameter.message || ""; var callback = e.parameter.callback || ""; if (message === "") { return response( { error: "Message is required" }, callback ); } var key = PropertiesService .getScriptProperties() .getProperty("GEMINI_API_KEY"); if (!key) { return response( { error: "API key not configured" }, callback ); } // Automatic fallback models var models = [ "gemini-3.8-flash", "gemini-3.7-flash", "gemini-3.6-flash", "gemini-3.5-flash-lite", "gemini-3.1-flash-lite", "gemini-2.5-flash-lite" ]; var lastError = ""; for (var i = 0; i < models.length; i++) { var model = models[i]; var url = "https://generativelanguage.googleapis.com/v1beta/models/" + model + ":generateContent"; var data = { contents: [ { parts: [ { text: message } ] } ] }; try { var result = UrlFetchApp.fetch(url, { method: "post", contentType: "application/json", headers: { "x-goog-api-key": key }, payload: JSON.stringify(data), muteHttpExceptions: true }); var status = result.getResponseCode(); var text = result.getContentText(); var json = JSON.parse(text); // Success if ( status >= 200 && status < 300 && json.candidates && json.candidates[0] && json.candidates[0].content && json.candidates[0].content.parts && json.candidates[0].content.parts[0] ) { var answer = json.candidates[0].content.parts[0].text; return response( { answer: answer, model: model }, callback ); } // Model failed — try next model if (json.error) { lastError = json.error.message; } else { lastError = "HTTP " + status; } } catch (err) { lastError = err.toString(); } } // All models failed return response( { error: "All Gemini models are temporarily unavailable. Please try again later. Last error: " + lastError }, callback ); } function response(data, callback) { var text = JSON.stringify(data); if (callback) { return ContentService .createTextOutput( callback + "(" + text + ")" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } return ContentService .createTextOutput(text) .setMimeType( ContentService.MimeType.JSON ); }
